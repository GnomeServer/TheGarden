#!/usr/bin/env bash
set -euo pipefail

cd -- "$(dirname -- "$0")"

read -rsp 'Agent Manager token: ' AGENT_MANAGER_API_TOKEN
printf '\n'

if [[ -n "${RUN_ID:-}" ]]; then
  : # Use RUN_ID when supplied by the caller.
elif [[ -f .last-run-id ]]; then
  RUN_ID=$(<.last-run-id)
else
  read -rp 'Run ID: ' RUN_ID
fi

curl -fsS \
  -H "Authorization: Bearer ${AGENT_MANAGER_API_TOKEN}" \
  "http://127.0.0.1:8090/v1/runs/${RUN_ID}"
printf '\n'
